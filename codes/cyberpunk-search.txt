            <input type="text" id="searchInput" placeholder="Кодҳоро кобед..." 
                class="search-input bg-transparent py-3 text-white placeholder-slate-600 text-sm focus:outline-none font-sans">
            
            <div id="searchBtn" class="w-[48px] h-[48px] flex items-center justify-center text-cyan-400 hover:text-purple-400 transition-colors duration-300 shrink-0">

    <script>
        const searchWrapper = document.getElementById('searchWrapper');
        const searchInput = document.getElementById('searchInput');
        const searchBtn = document.getElementById('searchBtn');

        // Вақте значокро пахш мекунад
        searchBtn.addEventListener('click', (e) => {
            e.stopPropagation(); // Барои он ки клик ба контейнер таъсир накунад
            searchWrapper.classList.toggle('active');
            
            if (searchWrapper.classList.contains('active')) {
                searchInput.focus();
            } else {
                searchInput.value = '';
            }
        });

        // Агар берун аз ҷустуҷӯ клик шавад, онро мепӯшонад
        document.addEventListener('click', (e) => {
            if (!searchWrapper.contains(e.target)) {
                searchWrapper.classList.remove('active');
                searchInput.value = '';
            }
        });
    </script>
